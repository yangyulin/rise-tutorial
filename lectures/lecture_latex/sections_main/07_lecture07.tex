\subsection{Preintegration}
\label{sec:lec07-preint}

Recall the IMU kinematics $\dot{\vect{x}}_I = \vect{f}(\vect{x}_I, \vect{n}_I)$ (Lecture~5). Integrating over one IMU interval gives
\begin{align}
\vect{x}_{k+1} = \vect{f}_{\mathrm{int}}(\vect{x}_k, \vect{n}_I)
\qquad\leftarrow\;\text{1 measurement: } k \to k+1,
\end{align}
while integrating across \emph{multiple} IMU measurements up to $t_j$ gives
\begin{align}
\vect{x}_j = \vect{f}_{\mathrm{int}}(\vect{x}_k, \vect{n}_I)
\qquad\leftarrow\;\text{multiple measurements: } k \to j.
\label{eq:lec07-multi-int}
\end{align}
By the exact integration results of Lecture~5, \eqref{eq:lec07-multi-int} can be described as (with $\delta t_k \triangleq t_j - t_k$)
\begin{align}
\begin{cases}
{}^{G}_{I_j}\vect{R} = {}^{G}_{I_k}\vect{R}\cdot\Delta\vect{R}, \\[2pt]
{}^{G}\vect{p}_{I_j} = {}^{G}\vect{p}_{I_k} + {}^{G}\vect{v}_{I_k}\,\delta t_k + {}^{G}_{I_k}\vect{R}\,\Delta\vect{p} + \frac{1}{2}{}^{G}\vect{g}\,\delta t_k^2, \\[2pt]
{}^{G}\vect{v}_{I_j} = {}^{G}\vect{v}_{I_k} + {}^{G}_{I_k}\vect{R}\,\Delta\vect{v} + {}^{G}\vect{g}\,\delta t_k, \\[2pt]
\vect{b}_{g_j} = \vect{b}_{g_k} + \Delta\vect{b}_g, \\[2pt]
\vect{b}_{a_j} = \vect{b}_{a_k} + \Delta\vect{b}_a.
\end{cases}
\label{eq:lec07-int-kj}
\end{align}
It is clear that $\Delta\vect{R}, \Delta\vect{p}$ and $\Delta\vect{v}$ are \emph{integration terms}, driven by the bias-corrected inertial readings
\begin{align}
{}^{I}\vect{a} = {}^{I}\vect{a}_m - \vect{b}_a - \vect{n}_a,
\qquad
{}^{I}\vect{\omega} = {}^{I}\vect{\omega}_m - \vect{b}_g - \vect{n}_g,
\label{eq:lec07-imumodel}
\end{align}
while $\Delta\vect{b}_g, \Delta\vect{b}_a$ are the integrated random walks. We can then reorganize the measurements:
\begin{align}
\Delta\vect{R} &\triangleq \Delta\vect{R}(\vect{b}_g, \vect{n}_I)
\triangleq {}^{I_k}_{I_j}\vect{R}
&&= \underbrace{{}^{G}_{I_k}\vect{R}\trans\,{}^{G}_{I_j}\vect{R}}_{\displaystyle \vect{h}_R(\vect{x}_k, \vect{x}_j)}, \\
\Delta\vect{p} &\triangleq \Delta\vect{p}(\vect{b}_a, \vect{b}_g, \vect{n}_I)
\triangleq \int_{t_k}^{t_j}\!\!\int_{t_k}^{\tau} {}^{I_k}_{I_s}\vect{R}\;{}^{I_s}\vect{a}\,ds\,d\tau
&&= \underbrace{{}^{G}_{I_k}\vect{R}\trans\Big({}^{G}\vect{p}_{I_j} - {}^{G}\vect{p}_{I_k} - {}^{G}\vect{v}_{I_k}\,\delta t_k - \tfrac{1}{2}{}^{G}\vect{g}\,\delta t_k^2\Big)}_{\displaystyle \vect{h}_p(\vect{x}_k, \vect{x}_j)}, \\
\Delta\vect{v} &\triangleq \Delta\vect{v}(\vect{b}_a, \vect{b}_g, \vect{n}_I)
\triangleq \int_{t_k}^{t_j} {}^{I_k}_{I_\tau}\vect{R}\;{}^{I_\tau}\vect{a}\,d\tau
&&= \underbrace{{}^{G}_{I_k}\vect{R}\trans\Big({}^{G}\vect{v}_{I_j} - {}^{G}\vect{v}_{I_k} - {}^{G}\vect{g}\,\delta t_k\Big)}_{\displaystyle \vect{h}_v(\vect{x}_k, \vect{x}_j)}, \\
\Delta\vect{b}_g &\triangleq \Delta\vect{b}_g(\vect{n}_I)
\triangleq \int_{t_k}^{t_j} \vect{n}_{wg}\,d\tau
&&= \vect{b}_{g_j} - \vect{b}_{g_k}, \\
\Delta\vect{b}_a &\triangleq \Delta\vect{b}_a(\vect{n}_I)
\triangleq \int_{t_k}^{t_j} \vect{n}_{wa}\,d\tau
&&= \vect{b}_{a_j} - \vect{b}_{a_k}.
\end{align}
Two sides of the same equations:
\begin{itemize}[noitemsep,topsep=2pt]
\item \textbf{Left-hand side:} the measurements from the IMU (relates to the IMU biases and noises);
\item \textbf{Right-hand side:} the nonlinear functions containing the states $\vect{x}$:
\begin{align}
\begin{cases}
\Delta\vect{R}(\vect{b}_g, \vect{n}_I) = \vect{h}_R(\vect{x}_k, \vect{x}_j), \\
\Delta\vect{p}(\vect{b}_a, \vect{b}_g, \vect{n}_I) = \vect{h}_p(\vect{x}_k, \vect{x}_j), \\
\Delta\vect{v}(\vect{b}_a, \vect{b}_g, \vect{n}_I) = \vect{h}_v(\vect{x}_k, \vect{x}_j), \\
\Delta\vect{b}_g(\vect{n}_I) = \vect{b}_{g_j} - \vect{b}_{g_k}, \\
\Delta\vect{b}_a(\vect{n}_I) = \vect{b}_{a_j} - \vect{b}_{a_k}
\end{cases}
\;\;\Rightarrow\;\;
\text{the IMU measurements, no assumption yet.}
\end{align}
\end{itemize}

\paragraph{Linearizing about the bias estimates.} Give an estimate of $\hat{\vect{b}}_g$ and $\hat{\vect{b}}_a$. Since $\Delta\vect{p}$ and $\Delta\vect{v}$ depend on \emph{both} biases, it is convenient to collect them into a single bias state and to stack the corresponding Jacobians:
\begin{align}
\vect{x}_b \triangleq \begin{bmatrix} \vect{b}_g \\ \vect{b}_a \end{bmatrix},
\qquad
\vect{H}_b^p \triangleq \begin{bmatrix} \vect{H}_g^p & \vect{H}_a^p \end{bmatrix},
\qquad
\vect{H}_b^v \triangleq \begin{bmatrix} \vect{H}_g^v & \vect{H}_a^v \end{bmatrix},
\label{eq:lec07-xb}
\end{align}
so that e.g.\ $\vect{H}_b^p\,\tilde{\vect{x}}_b = \vect{H}_g^p\,\tilde{\vect{b}}_g + \vect{H}_a^p\,\tilde{\vect{b}}_a$. ($\Delta\vect{R}$ depends on $\vect{b}_g$ only, so it keeps the single Jacobian $\vect{H}_g^R$.) To first order, then,
\begin{align}
\Delta\vect{R}(\vect{b}_g, \vect{n}_I)
&\approx \Delta\vect{R}(\hat{\vect{b}}_g, \vect{0})\cdot\Exp\big(\vect{H}_g^R\,\tilde{\vect{b}}_g + \vect{H}_n^R\,\vect{n}_I\big) \notag \\
\Rightarrow\quad
\Delta\vect{R}(\hat{\vect{b}}_g, \vect{0})
&= \vect{h}_R(\vect{x}_k, \vect{x}_j)\cdot\Exp\big(-\vect{H}_g^R\,\tilde{\vect{b}}_g - \vect{H}_n^R\,\vect{n}_I\big);
\end{align}
similarly,
\begin{align}
\Delta\vect{p}(\vect{b}_g, \vect{b}_a, \vect{n}_I)
&\approx \Delta\vect{p}(\hat{\vect{b}}_g, \hat{\vect{b}}_a, \vect{0})
+ \vect{H}_b^p\,\tilde{\vect{x}}_b + \vect{H}_n^p\,\vect{n}_I \notag \\
\Rightarrow\quad
\Delta\vect{p}(\hat{\vect{b}}_g, \hat{\vect{b}}_a, \vect{0})
&= \vect{h}_p(\vect{x}_k, \vect{x}_j)
- \vect{H}_b^p\,\tilde{\vect{x}}_b - \vect{H}_n^p\,\vect{n}_I, \\[4pt]
\Delta\vect{v}(\vect{b}_g, \vect{b}_a, \vect{n}_I)
&\approx \Delta\vect{v}(\hat{\vect{b}}_g, \hat{\vect{b}}_a, \vect{0})
+ \vect{H}_b^v\,\tilde{\vect{x}}_b + \vect{H}_n^v\,\vect{n}_I \notag \\
\Rightarrow\quad
\Delta\vect{v}(\hat{\vect{b}}_g, \hat{\vect{b}}_a, \vect{0})
&= \vect{h}_v(\vect{x}_k, \vect{x}_j)
- \vect{H}_b^v\,\tilde{\vect{x}}_b - \vect{H}_n^v\,\vect{n}_I.
\end{align}
Then, for the biases, we get
\begin{align}
\Delta\vect{b}_g(\vect{n}_I) = \vect{0} + \int_{t_k}^{t_j} \vect{n}_{wg}\,dt = \vect{0} + \vect{n}_{wgd}\,\delta t_k,
\qquad
\Delta\vect{b}_a(\vect{n}_I) = \vect{0} + \int_{t_k}^{t_j} \vect{n}_{wa}\,dt = \vect{0} + \vect{n}_{wad}\,\delta t_k,
\end{align}
with $\vect{n}_{wgd}, \vect{n}_{wad}$ the discrete noises. Hence \textbf{the cleaner IMU measurement} is:
\begin{empheq}[box=\fbox]{align}
\Delta\vect{R}(\hat{\vect{b}}_g, \vect{0})
&= \vect{h}_R(\vect{x}_k, \vect{x}_j)\cdot\Exp\big(-\vect{H}_g^R\,\tilde{\vect{b}}_g - \vect{H}_n^R\,\vect{n}_I\big), \notag \\
\Delta\vect{p}(\hat{\vect{b}}_g, \hat{\vect{b}}_a, \vect{0})
&= \vect{h}_p(\vect{x}_k, \vect{x}_j) - \vect{H}_b^p\,\tilde{\vect{x}}_b - \vect{H}_n^p\,\vect{n}_I, \notag \\
\Delta\vect{v}(\hat{\vect{b}}_g, \hat{\vect{b}}_a, \vect{0})
&= \vect{h}_v(\vect{x}_k, \vect{x}_j) - \vect{H}_b^v\,\tilde{\vect{x}}_b - \vect{H}_n^v\,\vect{n}_I, \notag \\
\vect{0} &= \vect{b}_{g_j} - \vect{b}_{g_k} - \vect{n}_{wgd}\,\delta t_k, \notag \\
\vect{0} &= \vect{b}_{a_j} - \vect{b}_{a_k} - \vect{n}_{wad}\,\delta t_k. \notag
\label{eq:lec07-cleaner}
\end{empheq}
Stacking the left-hand sides as the \emph{preintegrated measurement} $\vect{z}_I$, with $\vect{n}_I \sim \mathcal{N}(\vect{0}, \vect{R}_{ij})$:
\begin{align}
\vect{z}_I = \vect{h}_I(\vect{x}_k, \vect{x}_j, \vect{n}_I)
\;\;&\Rightarrow\;\;
\vect{z}_I = \underbrace{\vect{h}_I(\hat{\vect{x}}_k, \hat{\vect{x}}_j, \vect{0})}_{\hat{\vect{z}}_I}
+ \vect{H}_k\,\tilde{\vect{x}}_k + \vect{H}_j\,\tilde{\vect{x}}_j + \vect{G}_I\,\vect{n}_I \notag \\
&\Rightarrow\;\;
\tilde{\vect{z}}_I = \vect{H}_k\,\tilde{\vect{x}}_k + \vect{H}_j\,\tilde{\vect{x}}_j + \vect{G}_I\,\vect{n}_I.
\label{eq:lec07-imu-factor}
\end{align}

\paragraph{Why would we like this change?} Because this is exactly the format a least-squares solver wants: a residual that is linear in the state errors. Writing the covariance of the preintegrated measurement as
\begin{align}
\vect{Q}_I \triangleq \vect{G}_I\,\vect{R}_{ij}\,\vect{G}_I\trans,
\label{eq:lec07-QI}
\end{align}
we can form a \emph{cost term}
\begin{align}
\norm{\vect{z}_I - \vect{h}_I(\vect{x}_k, \vect{x}_j, \vect{0})}^2_{\vect{Q}_I}
\;\;\Rightarrow\;\;
\norm{\tilde{\vect{z}}_I - \vect{H}_k\,\tilde{\vect{x}}_k - \vect{H}_j\,\tilde{\vect{x}}_j}^2_{\vect{Q}_I},
\end{align}
which can be used in BA / least squares together with the vision factors easily.

But there is a \textbf{caveat} for the above formulation:
\begin{enumerate}[noitemsep,topsep=2pt]
\item for the least squares / MLE, we have multiple iterations;
\item for multi-iterations, $\hat{\vect{b}}_g$ is changing; then $\Delta\vect{p}, \Delta\vect{R}, \Delta\vect{v}$ and $\vect{G}_I, \vect{R}_{ij}$ will change;
\item if so, for every iteration we need to recompute $\vect{z}_I$ and $\vect{G}_I, \vect{R}_{ij}$;
\item if we have a high-frequency IMU and a lot of such IMU costs in the MLE, it will cause substantial computations.
\end{enumerate}

\paragraph{How to solve this?} The question is: \emph{what is the way to only integrate once and avoid repeated integration?} Consider a generic measurement $\vect{z} = \vect{h}(\vect{x}_A, \vect{x}_B) + \vect{n}$, and let $\hat{\vect{x}}_B^0$ be a \emph{fixed} linearization point for $\vect{x}_B$ --- the \emph{initial} estimate --- while $\hat{\vect{x}}_B$ is the current estimate. Writing the error with respect to the fixed point,
\begin{align}
\tilde{\vect{x}}_B^0 \triangleq \vect{x}_B - \hat{\vect{x}}_B^0
= \big(\vect{x}_B - \hat{\vect{x}}_B\big) + \big(\hat{\vect{x}}_B - \hat{\vect{x}}_B^0\big)
= \tilde{\vect{x}}_B + \Delta\hat{\vect{x}}_B,
\qquad \Delta\hat{\vect{x}}_B \triangleq \hat{\vect{x}}_B - \hat{\vect{x}}_B^0.
\label{eq:lec07-fixedlin-split}
\end{align}
Linearizing about $\hat{\vect{x}}_B^0$ --- so that $\vect{H}_B^0$ is evaluated at the initial estimate and never re-evaluated ---
\begin{align}
\tilde{\vect{z}} &= \vect{H}_A\,\tilde{\vect{x}}_A + \vect{H}_B^0\,\tilde{\vect{x}}_B^0 + \vect{n}
= \vect{H}_A\,\tilde{\vect{x}}_A + \vect{H}_B^0\,\tilde{\vect{x}}_B + \vect{H}_B^0\,\Delta\hat{\vect{x}}_B + \vect{n} \notag \\
\Rightarrow\quad
&\underbrace{\tilde{\vect{z}}}_{\text{unchanged}} - \boxed{\vect{H}_B^0\,\Delta\hat{\vect{x}}_B}
= \vect{H}_A\,\tilde{\vect{x}}_A + \underbrace{\vect{H}_B^0\,\tilde{\vect{x}}_B}_{\text{unchanged}} + \underbrace{\vect{n}}_{\text{unchanged}},
\label{eq:lec07-fixedlin}
\end{align}
where across iterations \emph{only the boxed term $\vect{H}_B^0\,\Delta\hat{\vect{x}}_B$ changes} --- a small computation: the Jacobians (and hence the preintegrated terms and noise mapping) stay at the fixed linearization point, and only the correction $\Delta\hat{\vect{x}}_B$ is updated.

\subsection{Preintegration with a Fixed Bias Linearization Point}
\label{sec:lec07-fixedbias}

Everything is now in place: the only change from \S\ref{sec:lec07-preint} is \emph{where} the bias Jacobians are evaluated. Instead of linearizing about the \emph{current} estimates $\hat{\vect{b}}_g, \hat{\vect{b}}_a$ --- which move at every Gauss--Newton / LM iteration and therefore force a re-integration --- we linearize about the \emph{fixed initial} estimates $\hat{\vect{b}}_g^0, \hat{\vect{b}}_a^0$ and pay for it with one extra correction term per row, exactly as in \eqref{eq:lec07-fixedlin}.

\paragraph{Applying the fixed linearization point.} Using the bias state $\vect{x}_b$ of \eqref{eq:lec07-xb}, define the corrections
\begin{align}
\Delta\hat{\vect{x}}_b \triangleq \begin{bmatrix} \hat{\vect{b}}_g - \hat{\vect{b}}_g^0 \\[2pt] \hat{\vect{b}}_a - \hat{\vect{b}}_a^0 \end{bmatrix},
\qquad
\Delta\hat{\vect{x}}_{b_g} \triangleq \hat{\vect{b}}_g - \hat{\vect{b}}_g^0 .
\label{eq:lec07-dxb}
\end{align}
For the rotation, splitting $\tilde{\vect{b}}_g^0 = \Delta\hat{\vect{x}}_{b_g} + \tilde{\vect{b}}_g$ exactly as in \eqref{eq:lec07-fixedlin-split},
\begin{align}
\Delta\vect{R}(\vect{b}_g, \vect{n}_I)
&= \Delta\vect{R}\big(\hat{\vect{b}}_g^0 + \tilde{\vect{b}}_g^0,\, \vect{n}_I\big)
\;\approx\; \Delta\vect{R}(\hat{\vect{b}}_g^0, \vect{0})\cdot\Exp\big(\vect{H}_g^R\,\tilde{\vect{b}}_g^0 + \vect{H}_n^R\,\vect{n}_I\big) \notag \\
&= \Delta\vect{R}(\hat{\vect{b}}_g^0, \vect{0})\cdot\Exp\big(\vect{H}_g^R\,\Delta\hat{\vect{x}}_{b_g} + \vect{H}_g^R\,\tilde{\vect{b}}_g + \vect{H}_n^R\,\vect{n}_I\big)
\;=\; \vect{h}_R(\vect{x}_k, \vect{x}_j) \notag \\
\Rightarrow\quad
\Delta\vect{R}(\hat{\vect{b}}_g^0, \vect{0})
&= \vect{h}_R(\vect{x}_k, \vect{x}_j)\cdot\Exp\big(-\vect{H}_g^R\,\Delta\hat{\vect{x}}_{b_g} - \vect{H}_g^R\,\tilde{\vect{b}}_g - \vect{H}_n^R\,\vect{n}_I\big).
\end{align}
Likewise for position and velocity, with the same stacked Jacobians $\vect{H}_b^p, \vect{H}_b^v$ of \eqref{eq:lec07-xb} now evaluated at the fixed point:
\begin{align}
\Delta\vect{p}(\vect{b}_g, \vect{b}_a, \vect{n}_I)
&= \Delta\vect{p}(\hat{\vect{b}}_g^0, \hat{\vect{b}}_a^0, \vect{0})
+ \vect{H}_b^p\,\Delta\hat{\vect{x}}_b + \vect{H}_b^p\,\tilde{\vect{x}}_b + \vect{H}_n^p\,\vect{n}_I
\;=\; \vect{h}_p(\vect{x}_k, \vect{x}_j) \notag \\
\Rightarrow\quad
\Delta\vect{p}(\hat{\vect{b}}_g^0, \hat{\vect{b}}_a^0, \vect{0})
&= \vect{h}_p(\vect{x}_k, \vect{x}_j) - \vect{H}_b^p\,\Delta\hat{\vect{x}}_b - \vect{H}_b^p\,\tilde{\vect{x}}_b - \vect{H}_n^p\,\vect{n}_I, \\[4pt]
\Delta\vect{v}(\vect{b}_g, \vect{b}_a, \vect{n}_I)
&= \Delta\vect{v}(\hat{\vect{b}}_g^0, \hat{\vect{b}}_a^0, \vect{0})
+ \vect{H}_b^v\,\Delta\hat{\vect{x}}_b + \vect{H}_b^v\,\tilde{\vect{x}}_b + \vect{H}_n^v\,\vect{n}_I
\;=\; \vect{h}_v(\vect{x}_k, \vect{x}_j) \notag \\
\Rightarrow\quad
\Delta\vect{v}(\hat{\vect{b}}_g^0, \hat{\vect{b}}_a^0, \vect{0})
&= \vect{h}_v(\vect{x}_k, \vect{x}_j) - \vect{H}_b^v\,\Delta\hat{\vect{x}}_b - \vect{H}_b^v\,\tilde{\vect{x}}_b - \vect{H}_n^v\,\vect{n}_I.
\end{align}

\paragraph{The final IMU measurement.} Stacking the five rows gives the preintegrated measurement. This is row-for-row the ``cleaner IMU measurement'' box of \S\ref{sec:lec07-preint}, with $\hat{\vect{b}}_g, \hat{\vect{b}}_a \to \hat{\vect{b}}_g^0, \hat{\vect{b}}_a^0$ and one extra $\vect{H}\,\Delta\hat{\vect{x}}$ term per row --- and those boxed terms are the \emph{only} ones that change from iteration to iteration:
\begin{align}
\begin{bmatrix}
\Delta\vect{R}(\hat{\vect{b}}_g^0, \vect{0}) \\[3pt]
\Delta\vect{p}(\hat{\vect{b}}_g^0, \hat{\vect{b}}_a^0, \vect{0}) \\[3pt]
\Delta\vect{v}(\hat{\vect{b}}_g^0, \hat{\vect{b}}_a^0, \vect{0}) \\[3pt]
\Delta\vect{b}_g \\[3pt]
\Delta\vect{b}_a
\end{bmatrix}
=
\begin{bmatrix}
\vect{h}_R(\vect{x}_k, \vect{x}_j)\cdot\Exp\big(-\boxed{\vect{H}_g^R\,\Delta\hat{\vect{x}}_{b_g}} - \vect{H}_g^R\,\tilde{\vect{b}}_g - \vect{H}_n^R\,\vect{n}_I\big) \\[3pt]
\vect{h}_p(\vect{x}_k, \vect{x}_j) - \boxed{\vect{H}_b^p\,\Delta\hat{\vect{x}}_b} - \vect{H}_b^p\,\tilde{\vect{x}}_b - \vect{H}_n^p\,\vect{n}_I \\[3pt]
\vect{h}_v(\vect{x}_k, \vect{x}_j) - \boxed{\vect{H}_b^v\,\Delta\hat{\vect{x}}_b} - \vect{H}_b^v\,\tilde{\vect{x}}_b - \vect{H}_n^v\,\vect{n}_I \\[3pt]
\vect{b}_{g_j} - \vect{b}_{g_k} \\[3pt]
\vect{b}_{a_j} - \vect{b}_{a_k}
\end{bmatrix}
\;\;\triangleq\;\;
\vect{z}_I = \vect{h}_I(\vect{x}_k, \vect{x}_j, \vect{n}_I).
\label{eq:lec07-zI}
\end{align}
The whole point is that
\begin{itemize}[noitemsep,topsep=2pt]
\item $\vect{z}_I$: we only need to \textbf{integrate once}, given $\hat{\vect{b}}_g^0, \hat{\vect{b}}_a^0$;
\item $\Delta\hat{\vect{x}}_b = \big[\,\hat{\vect{b}}_g - \hat{\vect{b}}_g^0;\ \hat{\vect{b}}_a - \hat{\vect{b}}_a^0\,\big]$ is the \textbf{bias linearization-change correction}, cheap to apply at each iteration.
\end{itemize}

\paragraph{The IMU factor.} We can now construct the IMU factor (IMU cost term)
\begin{empheq}[box=\fbox]{align}
C_I = \norm{\vect{z}_I - \vect{h}_I(\vect{x}_k, \vect{x}_j, \vect{0})}^2_{\vect{Q}_I}, \notag
\end{empheq}
where $\vect{z}_I$ is the IMU \emph{preintegrated measurement} and $\vect{Q}_I$ of \eqref{eq:lec07-QI} the covariance of the IMU \emph{preintegrated noise}.

\subsection{Computing the Preintegration: Recursive Formulation}
\label{sec:lec07-recursion}

The remaining question is how to actually perform the IMU preintegration and obtain
\begin{align}
\vect{z}_I, \quad \vect{Q}_I, \quad \vect{H}_b^p, \quad \vect{H}_b^v, \quad \vect{H}_g^R .
\end{align}
There are many IMU readings between $k$ and $j$ (Fig.~\ref{fig:lec07-imutimeline}); the way to solve this is to use a \emph{recursive formulation} over the individual IMU intervals $i \in [k, j]$.

\begin{figure}[H]
\centering
\begin{tikzpicture}[>=stealth]
  \draw[thick] (0,0) -- (10,0);
  \foreach \x in {0,0.9,1.8,2.7,3.6,4.2,6.0,6.9,7.8,8.5,10} {
    \draw[thick] (\x,-0.16) -- (\x,0.16);
  }
  \foreach \x/\lb in {0/0, 0.9/1, 1.8/2, 2.7/3, 6.0/i, 6.9/{i+1}} {

\paragraph{Recursion.} Propagating the preintegrated terms one IMU interval at a time,
\begin{align}
\begin{cases}
\Delta\vect{R}_{i+1} = \Delta\vect{R}_i \cdot \vect{R}_{i,i+1}, \\[2pt]
\Delta\vect{p}_{i+1} = \Delta\vect{p}_i + \Delta\vect{v}_i\,\delta t_i + \Delta\vect{R}_i\,\vect{p}_{i,i+1}, \\[2pt]
\Delta\vect{v}_{i+1} = \Delta\vect{v}_i + \Delta\vect{R}_i\,\vect{v}_{i,i+1}, \\[2pt]
\Delta\vect{b}_{g,i+1} = \Delta\vect{b}_{g,i} + \int_{t_i}^{t_{i+1}} \vect{n}_{wg}\,dt, \\[2pt]
\Delta\vect{b}_{a,i+1} = \Delta\vect{b}_{a,i} + \int_{t_i}^{t_{i+1}} \vect{n}_{wa}\,dt,
\end{cases}
\qquad
\begin{cases}
\vect{R}_{i,i+1} = \Exp\big(\hat{\vect{\omega}}_i\,\delta t_i\big), \\[2pt]
\vect{p}_{i,i+1} = \displaystyle\int_{t_i}^{t_{i+1}}\!\!\int_{t_i}^{\tau} {}^{I_i}_{I_s}\vect{R}\;{}^{I_s}\vect{a}\;ds\,d\tau, \\[2pt]
\vect{v}_{i,i+1} = \displaystyle\int_{t_i}^{t_{i+1}} {}^{I_i}_{I_\tau}\vect{R}\;{}^{I_\tau}\vect{a}\;d\tau,
\end{cases}
\label{eq:lec07-recursion}
\end{align}
which advances the preintegrated measurement $\vect{z}_{I,i} \to \vect{z}_{I,i+1}$. The interval quantities $\vect{p}_{i,i+1}, \vect{v}_{i,i+1}$ are evaluated either in closed form under the ACI$^2$ / CPI models~\cite{Yang2020ACI2,Eckenhoff2019IJRR} or a linear-in-time assumption on the readings, or numerically with RK4.

\paragraph{Linearized error and covariance propagation.} If we linearize the recursion, the preintegrated error state and its covariance propagate as
\begin{align}
\tilde{\vect{z}}_{I,i+1} &= \vect{\Phi}_{(i+1,i)}\,\tilde{\vect{z}}_{I,i} + \boxed{\vect{\Phi}_b}\,\tilde{\vect{x}}_{b_k} + \vect{G}_i\,\vect{n}_{Id}, \\
\vect{Q}_{I,i+1} &= \vect{\Phi}_{(i+1,i)}\,\vect{Q}_{I,i}\,\vect{\Phi}_{(i+1,i)}\trans + \vect{G}_i\,\vect{Q}_d\,\vect{G}_i\trans,
\end{align}
where $\vect{n}_{Id}$ is the discrete IMU noise
\begin{align}
\vect{n}_{Id} \sim \mathcal{N}\!\left(\vect{0},\;
\begin{bmatrix}
\sigma_g^2\vect{I}_3 & & & \\
& \sigma_a^2\vect{I}_3 & & \\
& & \sigma_{wg}^2\vect{I}_3 & \\
& & & \sigma_{wa}^2\vect{I}_3
\end{bmatrix}\frac{1}{\delta t_i}\right),
\end{align}
and the boxed $\vect{\Phi}_b$ is exactly the block that \textbf{contains the Jacobians} $\vect{H}_g^R, \vect{H}_b^p, \vect{H}_b^v$ needed in \eqref{eq:lec07-zI}. With this, the IMU preintegration factor is complete.

\subsection{Discussion}
\label{sec:lec07-discussion}

\begin{enumerate}[topsep=3pt]
\item \textbf{The key idea} is to \emph{control the linearization of the biases}; the assumption behind it is that the biases are \emph{slowly varying}, so that $\Delta\hat{\vect{x}}_b$ stays small and the first-order correction remains valid.
\item This preintegration idea can be applied to \textbf{any other temporal signal}, e.g.\ wheel encoders or rotor RPM.
\item This preintegration keeps the linearization of the \emph{biases}, but it can be extended to \textbf{more parameters} --- e.g.\ the IMU intrinsics (scale / skew parameters).
\item \textbf{How to use the IMU factor?} One must initialize the velocity and the biases (cf.\ the visual-inertial initialization of Lecture~6).
\end{enumerate}

\subsection{Optimization-Based VINS (Graph VINS)}
\label{sec:lec07-graphvins}

A typical optimization-based VINS combines a prior, the IMU factor, and the camera factors:
\begin{align}
\begin{cases}
\vect{x}_k \sim \mathcal{N}(\hat{\vect{x}}_k, \vect{P}_k), \\[2pt]
\dot{\vect{x}} = \vect{f}(\vect{x}, \vect{n}_I) \;\to\; \vect{x}_{k+1} = \vect{f}_{\mathrm{int}}(\vect{x}_k, \vect{n}_{I_k})
\;\;\Rightarrow\;\; \vect{z}_I = \vect{h}(\vect{x}_k, \vect{x}_{k+1}, \vect{n}_I), \\[2pt]
\vect{z}_{k+1} = \vect{h}(\vect{x}_{k+1}) + \vect{n}_{k+1},
\end{cases}
\end{align}
so that the cost is
\begin{empheq}[box=\fbox]{align}
C_{\mathrm{VINS}} = \underbrace{\norm{\vect{x}_k - \hat{\vect{x}}_k}^2_{\vect{P}_k}}_{\text{prior}}
+ \underbrace{\norm{\vect{z}_I - \vect{h}(\vect{x}_k, \vect{x}_{k+1}, \vect{0})}^2_{\vect{Q}_I}}_{C_{\mathrm{IMU}}}
+ \underbrace{\norm{\vect{z}_{k+1} - \vect{h}(\vect{x}_{k+1})}^2_{\vect{R}_{k+1}}}_{C_{\mathrm{cam}}}. \notag
\end{empheq}

\paragraph{Method 1: transfer the IMU readings from IMU1 onto IMU0.} Using the rigid extrinsics ${}^{I_0}\vect{p}_{I_1}$ and differentiating the rigid-body kinematics,
\begin{align}
{}^{G}\vect{p}_{I_1} &= {}^{G}\vect{p}_{I_0} + {}^{G}_{I_0}\vect{R}\;{}^{I_0}\vect{p}_{I_1}, \\
{}^{G}\dot{\vect{p}}_{I_1} &= {}^{G}\dot{\vect{p}}_{I_0} + {}^{G}_{I_0}\dot{\vect{R}}\;{}^{I_0}\vect{p}_{I_1}, \\
{}^{G}\vect{v}_{I_1} &= {}^{G}\vect{v}_{I_0} + {}^{G}_{I_0}\vect{R}\,\skewsym{{}^{I_0}\vect{\omega}}\;{}^{I_0}\vect{p}_{I_1}, \\
{}^{G}\dot{\vect{v}}_{I_1} &= {}^{G}\dot{\vect{v}}_{I_0} + {}^{G}_{I_0}\dot{\vect{R}}\,\skewsym{{}^{I_0}\vect{\omega}}\;{}^{I_0}\vect{p}_{I_1}
+ {}^{G}_{I_0}\vect{R}\,\skewsym{{}^{I_0}\dot{\vect{\omega}}}\;{}^{I_0}\vect{p}_{I_1}, \\
{}^{G}\vect{a}_{I_1} &= {}^{G}\vect{a}_{I_0}
+ {}^{G}_{I_0}\vect{R}\,\skewsym{{}^{I_0}\vect{\omega}}^2\;{}^{I_0}\vect{p}_{I_1}
+ {}^{G}_{I_0}\vect{R}\,\skewsym{{}^{I_0}\vect{\alpha}}\;{}^{I_0}\vect{p}_{I_1},
\end{align}
and hence, expressed in the second IMU's own frame,
\begin{align}
{}^{I_1}\vect{a}_{I_1} = {}^{I_1}_{I_0}\vect{R}\Big(
{}^{I_0}\vect{a}_{I_0}
+ \skewsym{{}^{I_0}\vect{\omega}}^2\;{}^{I_0}\vect{p}_{I_1}
+ \underbrace{\skewsym{{}^{I_0}\vect{\alpha}}\;{}^{I_0}\vect{p}_{I_1}}_{\text{needs }\vect{\alpha}}
\Big),
\qquad \vect{\alpha} \triangleq \dot{\vect{\omega}} .
\end{align}
The caveat is explicit in the last term: the \textbf{angular acceleration} $\vect{\alpha}$ is needed to do the transfer, and it is not directly measured by a gyroscope.

\paragraph{Method 2: keep IMU1 as its own state, tied by the rigid transform.} Instead of transferring readings, we relate the two states directly:
\begin{align}
\begin{cases}
{}^{G}_{I_1}\vect{T} = {}^{G}_{I_0}\vect{T}\cdot{}^{I_0}_{I_1}\vect{T}, \\[2pt]
{}^{G}\vect{v}_{I_1} = {}^{G}\vect{v}_{I_0} + {}^{G}_{I_0}\vect{R}\,\skewsym{{}^{I_0}\vect{\omega}}\;{}^{I_0}\vect{p}_{I_1}.
\end{cases}
\end{align}
Based on the rigid transform we can therefore represent $\vect{x}_{\mathrm{IMU}_1}$ with $\vect{x}_{\mathrm{IMU}_0}$:
\begin{align}
\vect{x}_{\mathrm{IMU}_0} =
\begin{bmatrix}
{}^{G}_{I_0}\vect{R} \\ {}^{G}\vect{p}_{I_0} \\ {}^{G}\vect{v}_{I_0} \\ {}^{I_0}\vect{b}_g \\ {}^{I_0}\vect{b}_a
\end{bmatrix},
\qquad
\vect{x}_{\mathrm{IMU}_1} =
\begin{bmatrix}
{}^{G}_{I_1}\vect{R} \\ {}^{G}\vect{p}_{I_1} \\ {}^{G}\vect{v}_{I_1} \\ {}^{I_1}\vect{b}_g \\ {}^{I_1}\vect{b}_a
\end{bmatrix},
\end{align}
The pose block of $\vect{x}_{\mathrm{IMU}_1}$ is fully determined by $\vect{x}_{\mathrm{IMU}_0}$ and the extrinsics, so the only extra \emph{auxiliary} state that has to be kept per additional IMU is
\begin{align}
\vect{x}_{I_1} = \begin{bmatrix} {}^{G}\vect{v}_{I_1} \\ {}^{I_1}\vect{b}_g \\ {}^{I_1}\vect{b}_a \end{bmatrix}.
\end{align}
The resulting graph (Fig.~\ref{fig:lec07-multiimugraph}) has a \emph{base} IMU chain (IMU0) carrying the navigation states plus one \emph{auxiliary} IMU factor chain per additional unit~\cite{Eckenhoff2021MIMC}.
